# Birth
on_birth_child = {
	on_actions = { awr_notify_birth }
}
awr_notify_birth = {
	effect = {
		every_player = {
			# Pinned mother
			if = {
				limit = {
					any_pinned_character = { this = scope:mother }
				}
				send_interface_message = {
					type = awr_message_birth
					title = awr_birth_mother_notable.t
					desc = awr_birth_mother_notable.desc
					left_icon = scope:mother
					right_icon = scope:child
				}
			}

			# Pinned father
			else_if = {
				limit = {
					any_pinned_character = { this = scope:father }
				}
				send_interface_message = {
					type = awr_message_birth
					title = awr_birth_father_notable.t
					desc = awr_birth_father_notable.desc
					left_icon = scope:father
					right_icon = scope:child
				}
			}

			# Relevant mother
			else_if = {
				limit = {
					OR = {
						dynasty = scope:child.dynasty
						awr_is_relevant = { TARGET = scope:mother }
					}
				}
				send_interface_message = {
					type = awr_message_birth
					title = awr_birth_mother.t
					desc = awr_birth_mother.desc
					left_icon = scope:mother
					right_icon = scope:child
				}
			}

			# Relevant father
			else_if = {
				limit = {
					OR = {
						dynasty = scope:father.dynasty
						awr_is_relevant = { TARGET = scope:father }
					}
				}
				send_interface_message = {
					type = awr_message_birth
					title = awr_birth_father.t
					desc = awr_birth_father.desc
					left_icon = scope:father
					right_icon = scope:child
				}
			}
		}
	}
}

# Death
on_death = {
	on_actions = { awr_notify_death }
}
awr_notify_death = {
	effect = {
		save_scope_as = dead
		every_player = {
			limit = {
				NOT = {
					OR = {
						scope:dead = this
						scope:dead = { is_councillor_of = prev }
						is_close_family_of = scope:dead
					}
				}
				OR = {
					dynasty = scope:dead.dynasty
					this = scope:dead.liege
					liege = scope:dead
					liege = scope:dead.liege
				}
			}
			send_interface_message = {
				type = awr_message_death
				title = awr_death.t
				desc = awr_death.desc
				left_icon = scope:dead
			}
		}
	}
}

# Marriage
on_marriage = {
	on_actions = { awr_notify_marriage }
}
awr_notify_marriage = {
	effect = {
		save_scope_as = married
		every_player = {
			limit = {
				NOT = {
					OR = {
						is_parent_of = scope:married
						is_parent_of = scope:spouse
						this = scope:married
						this = scope:spouse
					}
				}
			}

			# Pinned
			if = {
				limit = {
					any_pinned_character = {
						OR = {
							this = scope:married
							this = scope:spouse
						}
					}
				}
				send_interface_message = {
					type = awr_message_marriage
					title = awr_marriage_notable.t
					desc = awr_marriage_notable.desc
					left_icon = scope:married
					right_icon = scope:spouse
				}
			}

			# Relevant
			else_if = {
				limit = {
					OR = {
						awr_is_relevant = { TARGET = scope:married }
						awr_is_relevant = { TARGET = scope:spouse }
					}
				}
				send_interface_message = {
					type = awr_message_marriage
					title = awr_marriage.t
					desc = awr_marriage.desc
					left_icon = scope:married
					right_icon = scope:spouse
				}
			}
		}
	}
}
