# Birth
on_birth_child = {
	on_actions = { awr_notify_birth }
}
awr_notify_birth = {
	effect = {
		every_player = {
			limit = {
				OR = {
					dynasty = scope:child.dynasty
					dynasty = scope:mother.dynasty
					has_relation_friend = scope:mother
					has_relation_rival = scope:mother
					this = scope:mother.liege
					any_pinned_character = {
						this = scope:mother
					}
				}
			}
			send_interface_message = {
				type = awr_message_birth
				title = awr_birth_mother.t
				desc = awr_birth_mother.desc
				left_icon = scope:mother
				right_icon = scope:child
			}
		}
		every_player = {
			limit = {
				OR = {
					dynasty = scope:father.dynasty
					has_relation_friend = scope:father
					has_relation_rival = scope:father
					this = scope:father.liege
					any_pinned_character = {
						this = scope:father
					}
				}
			}
			send_interface_message = {
				type = awr_message_birth
				title = awr_birth_father.t
				desc = awr_birth_father.desc
				left_icon = scope:father
				right_icon = scope:child
			}
		}
	}
}

# Death
on_death = {
	on_actions = { awr_notify_death }
}
awr_notify_death = {
	effect = {
		save_scope_as = dead
		every_player = {
			limit = {
				# Not in council (already in game)
				NOT = {
					scope:dead = { is_councillor_of = prev }
				}
				OR = {
					# Of your dynasty
					dynasty = scope:dead.dynasty
					# Your friend
					has_relation_friend = scope:dead
					# Your rival
					has_relation_rival = scope:dead
					# Your vassal
					this = scope:dead.liege
					# Pinned (already in game)
				}
			}
			send_interface_message = {
				type = awr_message_death
				title = awr_death.t
				desc = awr_death.desc
				left_icon = scope:dead
			}
		}
	}
}

# Marriage
on_marriage = {
	on_actions = { awr_notify_marriage }
}
awr_notify_marriage = {
	effect = {
		save_scope_as = married
		every_player = {
			limit = {
				# Not yours
				NOT = {
					this = scope:married
					this = scope:spouse
				}
				OR = {
					# Of your dynasty
					dynasty = scope:married.dynasty
					dynasty = scope:spouse.dynasty
					# Your friend
					has_relation_friend = scope:married
					has_relation_friend = scope:spouse
					# Your rival
					has_relation_rival = scope:married
					has_relation_rival = scope:spouse
					# Your vassal
					this = scope:married.liege
					this = scope:spouse.liege
					# Pinned
					any_pinned_character = {
						this = scope:married
					}
					any_pinned_character = {
						this = scope:spouse
					}
				}
			}
			send_interface_message = {
				type = awr_message_marriage
				title = awr_marriage.t
				desc = awr_marriage.desc
				left_icon = scope:married
				right_icon = scope:spouse
			}
		}
	}
}
